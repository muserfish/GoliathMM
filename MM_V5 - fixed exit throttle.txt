        // Safe getter (various builds)
        private static double GetOrderQtySafe(Order o)
        {
            try
            {
                var t = o.GetType();
                var p = t.GetProperty("Quantity") ?? t.GetProperty("Qty") ?? t.GetProperty("Volume");
                if (p != null)
                {
                    var v = p.GetValue(o);
                    if (v != null) return Convert.ToDouble(v);
                }
            }
            catch { }
            return double.NaN;
        }

        private static int GetOrderQtyRounded(Order o)
        {
            double qty = GetOrderQtySafe(o);
            if (double.IsNaN(qty) || double.IsInfinity(qty)) return 0;
            return (int)Math.Round(qty);
        }

        // ================= Lifecycle =================
        protected override void OnRun()
        {
            state = new Dictionary<Side, SideState> { { Side.Buy, SideState.Idle }, { Side.Sell, SideState.Idle } };
            orderIdBySide = new Dictionary<Side, string> { { Side.Buy, null }, { Side.Sell, null } };
            lastPlaceReqTime = new Dictionary<Side, DateTime> { { Side.Buy, DateTime.MinValue }, { Side.Sell, DateTime.MinValue } };
            lastAckTime = new Dictionary<Side, DateTime> { { Side.Buy, DateTime.MinValue }, { Side.Sell, DateTime.MinValue } };
            workingPrice = new Dictionary<Side, double> { { Side.Buy, double.NaN }, { Side.Sell, double.NaN } };
            quoteOrderIds = new HashSet<string>(StringComparer.Ordinal);
            mids = new Queue<MidSample>();

            if (symbol == null || account == null || symbol.ConnectionId != account.ConnectionId)
            {
                Log("Pick Symbol & Account on the same connection.", StrategyLoggingLevel.Error);
                return;
            }

            symbol = Core.GetSymbol(symbol.CreateInfo());
            if (symbol == null) { Log("Symbol not found.", StrategyLoggingLevel.Error); return; }

            if (hedgeEnabled && hedgeSymbol != null)
            {
                try
                {
                    hedgeSymbol = Core.GetSymbol(hedgeSymbol.CreateInfo());
                    if (hedgeSymbol != null && hedgeSymbol.ConnectionId != account.ConnectionId)
                    {
                        Log("Hedge symbol on different connection — hedging disabled.", StrategyLoggingLevel.Trading);
                        hedgeSymbol = null;
                    }
                }
                catch { hedgeSymbol = null; }
            }

            symbol.NewQuote += OnQuote;
            Core.Instance.OrderAdded += OnOrderAdded;
            Core.Instance.OrderRemoved += OnOrderRemoved;

            Log($"Started {symbol.Name} on {account.Name}", StrategyLoggingLevel.Trading);

            AdoptExistingQuotes();
        }

        private void AdoptExistingQuotes()
        {
            try
            {
                foreach (var side in new[] { Side.Buy, Side.Sell })
                {
                    if (state[side] != SideState.Idle) continue;

                    var ours = Core.Instance?.Orders?
                        .Where(o => o != null && IsOurs(o) && o.Side == side)
                        .ToArray();

                    if (ours == null || ours.Length != 1) continue;

                    var live = ours[0];
                    int desired = ComputeSideContracts(side);
                    if (desired > 0 && GetOrderQtyRounded(live) != desired) continue;

                    quoteOrderIds.Add(live.Id);
                    orderIdBySide[side] = live.Id;
                    workingPrice[side] = live.Price;
                    lastAckTime[side] = DateTime.UtcNow;
                    state[side] = SideState.Working;
                    D($"INIT adopt side={side} id={live.Id} px={F2(live.Price)} qty={GetOrderQtyRounded(live)}");
                }
            }
            catch { }
        }

        protected override void OnStop()
        {
            try
            {
                if (symbol != null) symbol.NewQuote -= OnQuote;
                Core.Instance.OrderAdded -= OnOrderAdded;
                Core.Instance.OrderRemoved -= OnOrderRemoved;
                TryCancel(Side.Buy);
                TryCancel(Side.Sell);
                Log("Stopped", StrategyLoggingLevel.Trading);
            }
            catch { }
        }

        // =================== Main loop ===================
        private void OnQuote(Symbol s, Quote q)
        {
            var now = DateTime.UtcNow;

            // stale-feed resync
            if (lastQuoteSeen != DateTime.MinValue &&
                staleFeedMs > 0 &&
                (now - lastQuoteSeen).TotalMilliseconds > staleFeedMs)
            {
                ResyncOpenOrders();
                D($"SNAP resync: staleFeedMs={staleFeedMs}");
            }
            lastQuoteSeen = now;

            if ((now - lastLoop).TotalMilliseconds < refreshMs) return;
            lastLoop = now;

            if (s.Bid <= 0 || s.Ask <= 0) return;

            double tick = s.TickSize > 0 ? s.TickSize : 0.25;
            double mid = (s.Bid + s.Ask) * 0.5;

            // Live net position
            int liveNet = GetLiveNetPosition();

            // L1 DOM tilt
            double domImb = 0.0;
            if (useDomImbalance)
            {
                double bidSz = 0, askSz = 0;
                try { bidSz = Convert.ToDouble(q?.BidSize); } catch { }
                try { askSz = Convert.ToDouble(q?.AskSize); } catch { }
                double denom = bidSz + askSz;
                if (denom > 0) domImb = (bidSz - askSz) / denom;

                double ms = Math.Max(50.0, refreshMs);
                double alpha = Clamp(2.0 / (Math.Max(1.0, domImbEmaSec) * (1000.0 / ms) + 1.0), 0.02, 0.95);
                domImbEma = domImbEma + alpha * (domImb - domImbEma);
            }

            // keep mid window
            PushMid(now, mid, TimeSpan.FromSeconds(Math.Max(5, volWindowSec)));
            int volTicks = EstimateVolatilityTicks(tick);
            int volBoostTicks = Math.Min(maxVolBoostTicks, (int)Math.Ceiling(volTicks * volBoost));

            // === Effective tick value (known map or manual) ===
            double effTickValue = tickDollarValue;
            if (useKnownTickMap)
            {
                double guessed = GuessTickValueByName(s?.Name);
                if (guessed > 0) effTickValue = guessed;
            }

            // commission-aware half-spread floor
            double ticksBE = effTickValue > 0 ? (2.0 * commissionPerSide) / effTickValue : 0.0;
            int beHalfFloor = (int)Math.Ceiling(0.5 * ticksBE) + beBufferTicks;
            int halfTicks = Math.Max(baseHalfTicks, beHalfFloor) + volBoostTicks;

            // simple trend
            var (tslope, tr2) = TrendSlopeTicksPerSec(tick);
            int trendDir = (Math.Abs(tslope) >= slopeThresh && tr2 >= trendR2Min) ? Math.Sign(tslope) : 0;

            // inventory skew
            int invSkewTicks = 0;
            if (enableInventory && invBand > 0)
            {
                if (Math.Abs(liveNet) >= invCap)
                    invSkewTicks = liveNet > 0 ? +invMaxSkewTicks : -invMaxSkewTicks;
                else
                {
                    int sign = Math.Sign(liveNet);
                    int mag = Math.Max(0, Math.Abs(liveNet) - invBand);
                    invSkewTicks = Clamp(sign * mag, -invMaxSkewTicks, +invMaxSkewTicks);
                }
            }

            // DOM skew
            int domSkewTicks = 0;
            if (useDomImbalance && Math.Abs(domImbEma) >= domImbThresh)
            {
                double scale = (Math.Abs(domImbEma) - domImbThresh) / Math.Max(1e-9, (1.0 - domImbThresh));
                int baseSkew = (int)Math.Min(domMaxSkewTicks, Math.Ceiling(scale * domMaxSkewTicks));
                domSkewTicks = (domImbEma >= 0) ? +baseSkew : -baseSkew;
            }

            // blend inv + DOM
            int blendedSkew = (int)Math.Round(invSkewTicks * (1.0 - domWeight) + domSkewTicks * domWeight);

            // exit-only tiers
            bool trendAgainst = (trendDir > 0 && liveNet < 0) || (trendDir < 0 && liveNet > 0);
            int outside = Math.Max(0, Math.Abs(liveNet) - invBand);
            bool tier1 = trendAgainst || outside >= tier1PastBand;
            bool tier2 = outside >= tier2PastBand;

            bool exitOnly = tier1 || (exitOnlySince != DateTime.MinValue && (now - exitOnlySince).TotalMilliseconds < exitOnlyHoldMs);
            if (exitOnly && exitOnlySince == DateTime.MinValue) exitOnlySince = now;
            if (!exitOnly && outside == 0) exitOnlySince = DateTime.MinValue;

            // NEW: reset IOC burst counter on enter/exit of exit-only
            if (exitOnly && !prevExitOnly) iocThisBurst = 0;
            if (!exitOnly && prevExitOnly) iocThisBurst = 0;
            prevExitOnly = exitOnly;

            // dynamic resting from vol
            double volNorm = 0.0;
            if (volRefHigh > volRefLow)
                volNorm = Clamp((double)(volTicks - volRefLow) / (volRefHigh - volRefLow), 0.0, 1.0);

            int dynRestBandTicks = (int)Math.Round(restBandMin + (restBandMax - restBandMin) * volNorm);
            int dynMinRestMs = (int)Math.Round(minRestSlowMs - (minRestSlowMs - minRestFastMs) * volNorm);

            // NEW: smooth main-quote churn while exit-only (soften, not a hard stop)
            if (exitOnly)
            {
                dynRestBandTicks += Math.Max(0, exitOnlyExtraBandTicks);
                dynMinRestMs += Math.Max(0, exitOnlyExtraRestMs);
            }

            // target quotes (mid anchor + skew)
            int bidTicksAdj = +blendedSkew;
            int askTicksAdj = -blendedSkew;
            int bidTicks = Math.Max(1, halfTicks + queueBehindTicks + bidTicksAdj);
            int askTicks = Math.Max(1, halfTicks + queueBehindTicks + askTicksAdj);

            double buyPx = RoundToTick(mid - bidTicks * tick, tick);
            double sellPx = RoundToTick(mid + askTicks * tick, tick);

            // caps & exit anchoring
            bool blockBuy = enableInventory && liveNet >= invCap;
            bool blockSell = enableInventory && liveNet <= -invCap;

            Side exitSide = (liveNet < 0) ? Side.Buy : (liveNet > 0) ? Side.Sell : Side.Buy;

            if (exitOnly)
            {
                // tighten exit anchoring
                if (exitSide == Side.Buy)
                {
                    if (symbol.Bid > 0) buyPx = RoundToTick(symbol.Bid - minInsideOffset * tick, tick);
                    blockSell = true;
                }
                else
                {
                    if (symbol.Ask > 0) sellPx = RoundToTick(symbol.Ask + minInsideOffset * tick, tick);
                    blockBuy = true;
                }

                // === Tier-2 extras (JOIN + IOC) with throttles ===
                if (tier2)
                {
                    // Guarded touch-join passive (cooldown + dedup + max clip count)
                    int joinQty = Math.Max(1, (int)Math.Round(baseContracts * touchJoinPct));
                    double joinPx = (exitSide == Side.Buy)
                        ? RoundToTick(symbol.Bid, tick)
                        : RoundToTick(symbol.Ask, tick);

                    bool cooldownOk = (now - lastPassiveJoinTime).TotalMilliseconds >= touchJoinCooldownMs;

                    int passiveOnSide = Core.Instance.Orders.Count(o =>
                        o != null && IsOurs(o) && o.Side == exitSide && o.OrderTypeId == OrderType.Limit);

                    bool alreadyJoinedAtPx = Core.Instance.Orders.Any(o =>
                        o != null && IsOurs(o) && o.Side == exitSide && o.OrderTypeId == OrderType.Limit
                        && Math.Abs(o.Price - joinPx) < tick * 0.5);

                    if (joinQty > 0 && cooldownOk && passiveOnSide < Math.Max(0, maxExitPassiveClips) && !alreadyJoinedAtPx)
                    {
                        PlacePassiveClip(exitSide, joinPx, joinQty);
                        lastPassiveJoinTime = now;
                        D($"TIER join: side={exitSide} px={F2(joinPx)} qty={joinQty} existingClips={passiveOnSide}");
                    }

                    // IOC nibble with per-burst soft cap
                    if ((now - lastTierStart).TotalMilliseconds >= tierEscalateMs && iocThisBurst < Math.Max(0, maxIocPerBurst))
                    {
                        int nibble = Math.Max(1, (int)Math.Ceiling(Math.Abs(liveNet) * iocNibblePct));
                        PlaceIoc(exitSide, nibble);
                        iocThisBurst++;
                        lastTierStart = now;
                        D($"TIER escalate: side={exitSide} nibble={nibble} iocBurst={iocThisBurst}/{maxIocPerBurst}");
                    }
                }
                else
                {
                    lastTierStart = DateTime.MinValue;
                }
            }
            else
            {
                lastTierStart = DateTime.MinValue;
            }

            // scaling
            int buyQty = ComputeSideContracts(Side.Buy);
            int sellQty = ComputeSideContracts(Side.Sell);

            // SNAP diag
            D($"SNAP mid={F2(mid)} bid={F2(s.Bid)} ask={F2(s.Ask)} " +
              $"inv={Sgn(liveNet)} exitOnly={Yn(exitOnly)} " +
              $"restBand={dynRestBandTicks} minRestMs={dynMinRestMs} " +
              $"tier2={Yn(tier2)} iocBurst={iocThisBurst}/{maxIocPerBurst} " +
              $"buyPx={F2(buyPx)} sellPx={F2(sellPx)} buyQty={buyQty} sellQty={sellQty}");

            // step each side
            StepSide(Side.Buy, buyPx, now, tick, blockBuy, buyQty, dynRestBandTicks, dynMinRestMs);
            StepSide(Side.Sell, sellPx, now, tick, blockSell, sellQty, dynRestBandTicks, dynMinRestMs);

            // hedge
            if (hedgeEnabled && hedgeSymbol != null)
                ManageHedge(now);
        }

        private void StepSide(
            Side side, double targetPrice, DateTime now, double tick, bool blockNewPlaces, int contractsForThisSide,
            int dynRestBandTicks, int dynMinRestMs)
        {
            double sinceAck = (now - lastAckTime[side]).TotalMilliseconds;
            double sincePlace = (now - lastPlaceReqTime[side]).TotalMilliseconds;

            bool allowAdopt = state[side] == SideState.Idle || state[side] == SideState.AwaitingAck;
            var live = GetQuoteOrder(side, allowAdopt, contractsForThisSide);

            if (state[side] == SideState.AwaitingAck && live != null)
            {
                state[side] = SideState.Working;
                lastAckTime[side] = DateTime.UtcNow;
                D($"STEP {side} fallback->Working id={live.Id} px={F2(live.Price)} qty={GetOrderQtyRounded(live)}");
            }

            if (usePollingAckFallback && state[side] == SideState.AwaitingAck && sincePlace > ackTimeoutMs / 2)
            {
                if (live == null && sincePlace > ackTimeoutMs && !blockNewPlaces)
                {
                    D($"STEP {side} re-place tgt={F2(targetPrice)} qty={contractsForThisSide}");
                    Place(side, targetPrice, contractsForThisSide);
                    lastPlaceReqTime[side] = DateTime.UtcNow;
                    return;
                }
            }

            bool forceRequote = (forceRequoteMs > 0) && (sinceAck >= forceRequoteMs);

            switch (state[side])
            {
                case SideState.Idle:
                    if (live != null)
                    {
                        lastAckTime[side] = DateTime.UtcNow;
                        state[side] = SideState.Working;
                        D($"STEP {side} adopt live id={live.Id} px={F2(live.Price)} qty={GetOrderQtyRounded(live)}");
                        break;
                    }

                    if (!blockNewPlaces)
                    {
                        D($"STEP {side} PLACE tgt={F2(targetPrice)} qty={contractsForThisSide}");
                        Place(side, targetPrice, contractsForThisSide);
                        state[side] = SideState.AwaitingAck;
                        lastPlaceReqTime[side] = DateTime.UtcNow;
                        lastAckTime[side] = DateTime.UtcNow;
                    }
                    break;

                case SideState.AwaitingAck:
                    if (blockNewPlaces)
                    {
                        D($"STEP {side} CANCEL (blocked awaiting ack)");
                        state[side] = SideState.Cancelling;
                        TryCancel(side);
                        lastAckTime[side] = DateTime.UtcNow;
                    }
                    break;

                case SideState.Working:
                {
                    if (live == null)
                    {
                        orderIdBySide[side] = null;
                        workingPrice[side] = double.NaN;
                        state[side] = SideState.Idle;
                        break;
                    }

                    if (blockNewPlaces)
                    {
                        D($"STEP {side} CANCEL (blocked) cur={F2(live.Price)}");
                        state[side] = SideState.Cancelling;
                        TryCancel(side);
                        lastAckTime[side] = DateTime.UtcNow;
                        break;
                    }

                    int liveQty = GetOrderQtyRounded(live);
                    if (contractsForThisSide > 0 && liveQty != contractsForThisSide)
                    {
                        D($"STEP {side} CANCEL qty={liveQty} -> target={contractsForThisSide}");
                        state[side] = SideState.Cancelling;
                        TryCancel(side);
                        lastAckTime[side] = DateTime.UtcNow;
                        break;
                    }

                    double cur = live.Price;
                    int deltaTicks = (int)Math.Abs(Math.Round((targetPrice - cur) / tick));
                    bool movedBeyondBand = deltaTicks >= dynRestBandTicks;
                    bool oldEnough = (now - lastAckTime[side]).TotalMilliseconds >= dynMinRestMs;

                    if ((movedBeyondBand && oldEnough) || forceRequote)
                    {
                        D($"STEP {side} CANCEL cur={F2(cur)} -> tgt={F2(targetPrice)} Δticks={deltaTicks}/{dynRestBandTicks}");
                        state[side] = SideState.Cancelling;
                        TryCancel(side);
                        lastAckTime[side] = DateTime.UtcNow;
                    }
                    break;
                }

                case SideState.Cancelling:
                    if ((now - lastAckTime[side]).TotalMilliseconds > ackTimeoutMs)
                    {
                        D($"STEP {side} CANCEL retry");
                        TryCancel(side);
                        lastAckTime[side] = DateTime.UtcNow;
                    }
                    break;
            }
        }

        }

        private void TryCancel(Side side)
        {
            try
            {
                var id = orderIdBySide[side];
                var ord = !string.IsNullOrEmpty(id) ? Core.Instance.GetOrderById(id, symbol.ConnectionId) : null;
                if (ord != null)
                {
                    Core.Instance.CancelOrder((IOrder)ord);
                    return;
                }

                var tracked = Core.Instance.Orders?
                    .Where(o => o != null && IsOurs(o) && o.Side == side && quoteOrderIds.Contains(o.Id))
                    .ToArray();

                if (tracked != null && tracked.Length > 0)
                {
                    foreach (var o in tracked) Core.Instance.CancelOrder((IOrder)o);
                    return;
                }

                var fallback = Core.Instance.Orders?
                    .Where(o => o != null && IsOurs(o) && o.Side == side)
                    .ToArray();

                if (fallback != null)
                    foreach (var o in fallback) Core.Instance.CancelOrder((IOrder)o);
            }
            catch { }
        }

        // === Hedge (incremental) ===
        private void ManageHedge(DateTime now)
        {
            if (hedgeSymbol == null) { D("HEDGE skip: hedgeSymbol=null"); return; }
            if (hedgeSymbol.ConnectionId != account.ConnectionId) { D("HEDGE skip: diff connection"); return; }

            int liveNet = GetLiveNetPosition();
            int absNet = Math.Abs(liveNet);

            if (absNet < hedgeThreshold) { D($"HEDGE skip: absNet={absNet} < thresh={hedgeThreshold}"); return; }
            if ((now - lastHedgeTime).TotalMilliseconds < hedgeCooldownMs) { D("HEDGE skip: cooldown"); return; }

            double ratio = Math.Max(1e-9, hedgeRatio);
            int desiredUnits = (int)Math.Floor((absNet - hedgeThreshold) / ratio) + 1;
            desiredUnits = Math.Min(Math.Max(1, desiredUnits), Math.Max(1, hedgeMax));

            int targetSign = (liveNet < 0) ? +1 : (liveNet > 0) ? -1 : 0;
            if (hedgeInvertSide) targetSign = -targetSign;

            int targetHedge = desiredUnits * targetSign;
            int curHedge = GetHedgeNetPosition();
            int delta = targetHedge - curHedge;

            D($"HEDGE check: base={Sgn(liveNet)} absNet={absNet} thresh={hedgeThreshold} ratio={hedgeRatio} desiredUnits={desiredUnits} target={Sgn(targetHedge)} current={Sgn(curHedge)} delta={Sgn(delta)} max={hedgeMax}");

            if (delta == 0) { D("HEDGE ok: already at target"); return; }

            double step = hedgeSymbol.LotStep > 0 ? hedgeSymbol.LotStep : 1.0;
            double minLot = hedgeSymbol.MinLot > 0 ? hedgeSymbol.MinLot : 1.0;
            double qtyAbs = Math.Max(minLot, Math.Round(Math.Abs(delta) / step) * step);

            Side side = (delta > 0) ? Side.Buy : Side.Sell;

            var req = new PlaceOrderRequestParameters
            {
                Account = account,
                Symbol = hedgeSymbol,
                Side = side,
                Quantity = qtyAbs,
                TimeInForce = TimeInForce.IOC,
                OrderTypeId = OrderType.Market
            };

            Core.Instance.PlaceOrder(req);
            lastHedgeTime = now;

            D($"HEDGE place: side={side} qty={qtyAbs} hedgeSym={hedgeSymbol?.Name} (base={Sgn(liveNet)} absNet={absNet} desired={Sgn(targetHedge)} current={Sgn(curHedge)} delta={Sgn(delta)})");
        }

        // === Positions (by Id — robust) ===
        private int GetLiveNetPosition()
        {
            try
            {
                var gp = Core.Instance?.Positions?.FirstOrDefault(p =>
                    p != null && AccountsEqual(p.Account, this.account) && SymbolsEqual(p.Symbol, this.symbol));
                if (gp != null) return (int)Math.Round(gp.Quantity);
                else if (debugLogs) D("POS base: no matching position (by Id) — fallback to manual netPosition.");
            }
            catch (Exception ex) { if (debugLogs) D($"POS base: exception {ex.Message}"); }
            return this.netPosition;
        }

        private int GetHedgeNetPosition()
        {
            try
            {
                var hp = Core.Instance?.Positions?.FirstOrDefault(p =>
                    p != null && AccountsEqual(p.Account, this.account) &&
                    this.hedgeSymbol != null && SymbolsEqual(p.Symbol, this.hedgeSymbol));
                if (hp != null) return (int)Math.Round(hp.Quantity);
                else if (debugLogs) D("POS hedge: no matching hedge position (by Id).");
            }
            catch (Exception ex) { if (debugLogs) D($"POS hedge: exception {ex.Message}"); }
            return 0;
        }

        private int ComputeSideContracts(Side side)
        {
            int liveNet = GetLiveNetPosition();
            int sign = Math.Sign(liveNet);
            bool isExitSide = (sign > 0 && side == Side.Sell) || (sign < 0 && side == Side.Buy);

            if (!isExitSide)
                return Math.Clamp(baseContracts, 1, maxContracts);

            int outside = Math.Max(0, Math.Abs(liveNet) - invBand);
            int scaled = baseContracts + outside * Math.Max(0, scaleQtySlope);
            return Math.Clamp(scaled, 1, maxContracts);
        }

        // === Events ===
        private void OnOrderAdded(Order o)
        {
            if (!IsOurs(o)) return;
            if (state[o.Side] != SideState.AwaitingAck) return;

            orderIdBySide[o.Side] = o.Id;
            quoteOrderIds.Add(o.Id);
            workingPrice[o.Side] = o.Price;
            lastAckTime[o.Side] = DateTime.UtcNow;
            state[o.Side] = SideState.Working;
            D($"ORD add side={o.Side} id={o.Id} px={F2(o.Price)} qty={GetOrderQtyRounded(o)}");
        }

        private void OnOrderRemoved(Order o)
        {
            if (!IsOurs(o)) return;
            quoteOrderIds.Remove(o.Id);
            if (orderIdBySide[o.Side] == o.Id)
            {
                orderIdBySide[o.Side] = null;
                workingPrice[o.Side] = double.NaN;
                lastAckTime[o.Side] = DateTime.UtcNow;
                state[o.Side] = SideState.Idle;
                D($"ORD rm side={o.Side} id={o.Id} -> state=Idle");
            }
        }

        private bool IsOurs(Order o)
        {
            if (o == null) return false;
            if (!AccountsEqual(o.Account, account) || !SymbolsEqual(o.Symbol, symbol)) return false;

            bool sawMetadata = false;

            try
            {
                var type = o.GetType();

                foreach (var name in new[] { "Strategy", "OwnerStrategy", "StrategyObject" })
                {
                    var prop = type.GetProperty(name);
                    if (prop == null) continue;
                    sawMetadata = true;

                    var val = prop.GetValue(o);
                    if (val == null) continue;

                    if (ReferenceEquals(val, this)) return true;
                    return false;
                }

                foreach (var name in new[] { "StrategyId", "StrategyID", "StrategyGuid" })
                {
                    var prop = type.GetProperty(name);
                    if (prop == null) continue;
                    sawMetadata = true;

                    var val = prop.GetValue(o)?.ToString();
                    if (string.IsNullOrEmpty(val)) continue;

                    var myIdProp = this.GetType().GetProperty("Id");
                    var myId = myIdProp?.GetValue(this)?.ToString();
                    if (!string.IsNullOrEmpty(myId))
                        return string.Equals(val, myId, StringComparison.OrdinalIgnoreCase);

                    return false;
                }

                foreach (var name in new[] { "StrategyName", "Strategy" })
                {
                    var prop = type.GetProperty(name);
                    if (prop == null || prop.PropertyType != typeof(string)) continue;
                    sawMetadata = true;

                    var val = prop.GetValue(o) as string;
                    if (string.IsNullOrEmpty(val)) continue;

                    if (!string.IsNullOrEmpty(this.Name) && string.Equals(val, this.Name, StringComparison.OrdinalIgnoreCase))
                        return true;

                    return false;
                }
            }
            catch { }

            return !sawMetadata;
        }

        private static bool AccountsEqual(Account a, Account b)
        {
            if (a == null || b == null) return false;
            if (ReferenceEquals(a, b)) return true;

            try
            {
                if (a.Id != null && b.Id != null && a.Id.Equals(b.Id)) return true;
            }
            catch { }

            if (!string.IsNullOrEmpty(a.Name) && !string.IsNullOrEmpty(b.Name) &&
                string.Equals(a.Name, b.Name, StringComparison.OrdinalIgnoreCase))
                return true;

            return false;
        }

        private static bool SymbolsEqual(Symbol a, Symbol b)
        {
            if (a == null || b == null) return false;
            if (ReferenceEquals(a, b)) return true;

            try
            {
                if (a.Id != null && b.Id != null && a.Id.Equals(b.Id)) return true;
            }
            catch { }

            if (!string.IsNullOrEmpty(a.Name) && !string.IsNullOrEmpty(b.Name) &&
                string.Equals(a.Name, b.Name, StringComparison.OrdinalIgnoreCase))
                return true;

            return false;
        }

        private Order GetQuoteOrder(Side side, bool allowAdopt, int expectedQty)
        {
            Order live = null;

            try
            {
                var id = orderIdBySide[side];
                if (!string.IsNullOrEmpty(id))
                {
                    try
                    {
                        live = Core.Instance.GetOrderById(id, symbol?.ConnectionId);
                        if (live == null || !IsOurs(live) || live.Side != side)
                        {
                            quoteOrderIds.Remove(id);
                            orderIdBySide[side] = null;
                            workingPrice[side] = double.NaN;
                            live = null;
                        }
                    }
                    catch
                    {
                        quoteOrderIds.Remove(id);
                        orderIdBySide[side] = null;
                        workingPrice[side] = double.NaN;
                    }
                }

                if (live == null)
                {
                    var tracked = Core.Instance.Orders?
                        .Where(o => o != null && IsOurs(o) && o.Side == side && quoteOrderIds.Contains(o.Id))
                        .ToArray();

                    if (tracked != null && tracked.Length == 1)
                        live = tracked[0];
                }

                if (live == null && allowAdopt)
                {
                    var ours = Core.Instance.Orders?
                        .Where(o => o != null && IsOurs(o) && o.Side == side)
                        .ToArray();

                    if (ours != null && ours.Length == 1)
                    {
                        var candidate = ours[0];
                        int qty = GetOrderQtyRounded(candidate);
                        if (expectedQty <= 0 || qty == expectedQty)
                        {
                            live = candidate;
                            quoteOrderIds.Add(live.Id);
                        }
                    }
                }

                if (live != null)
                {
                    orderIdBySide[side] = live.Id;
                    workingPrice[side] = live.Price;
                }
            }
            catch { }

            return live;
        }

        private void PushMid(DateTime t, double price, TimeSpan window)
        {
            if (mids == null) mids = new Queue<MidSample>();
            mids.Enqueue(new MidSample { Time = t, Price = price });
            var cutoff = t - window;
            while (mids.Count > 0 && mids.Peek().Time < cutoff) mids.Dequeue();
        }

        private int EstimateVolatilityTicks(double tick)
        {
            if (mids == null || mids.Count < 2 || tick <= 0) return 0;
            double sum = 0; int n = 0;
            MidSample prev = mids.Peek();
            foreach (var x in mids)
            {
                if (x.Time == prev.Time) continue;
                double dticks = Math.Abs((x.Price - prev.Price) / tick);
                sum += dticks; n++;
                prev = x;
            }
            return n == 0 ? 0 : (int)Math.Round(sum / n);
        }

        private (double slope, double r2) TrendSlopeTicksPerSec(double tick)
        {
            if (mids == null) return (0, 0);
            var now = DateTime.UtcNow;
            var cut = now - TimeSpan.FromSeconds(Math.Max(10, trendWindowSec));
            var xs = new List<double>();
            var ys = new List<double>();
            foreach (var m in mids)
            {
                if (m.Time < cut) continue;
                xs.Add((m.Time - cut).TotalSeconds);
                ys.Add(m.Price / Math.Max(tick, 1e-9));
            }
            int n = xs.Count; if (n < 5) return (0, 0);
            double xm = 0, ym = 0;
            for (int i = 0; i < n; i++) { xm += xs[i]; ym += ys[i]; }
            xm /= n; ym /= n;
            double num = 0, den = 0, varY = 0;
            for (int i = 0; i < n; i++) { double dx = xs[i] - xm, dy = ys[i] - ym; num += dx * dy; den += dx * dx; varY += dy * dy; }
            if (den <= 0 || varY <= 0) return (0, 0);
            double slope = num / den; // ticks/sec
            double r2 = (num * num) / (den * varY);
            return (slope, r2);
        }

        private static int Clamp(int v, int lo, int hi) => v < lo ? lo : (v > hi ? hi : v);
        private static double Clamp(double v, double lo, double hi) => v < lo ? lo : (v > hi ? hi : v);

        private double GuessTickValueByName(string name)
        {
            if (string.IsNullOrEmpty(name)) return -1;
            foreach (var kv in KnownTickValues)
                if (name.StartsWith(kv.Key, StringComparison.OrdinalIgnoreCase))
                    return kv.Value;
            foreach (var kv in KnownTickValues)
                if (name.IndexOf(kv.Key, StringComparison.OrdinalIgnoreCase) >= 0)
                    return kv.Value;
            return -1;
        }

        private void ResyncOpenOrders()
        {
            TryCancel(Side.Buy);
            TryCancel(Side.Sell);
            orderIdBySide[Side.Buy] = null;
            orderIdBySide[Side.Sell] = null;
            workingPrice[Side.Buy] = double.NaN;
            workingPrice[Side.Sell] = double.NaN;
            lastAckTime[Side.Buy] = DateTime.UtcNow;
            lastAckTime[Side.Sell] = DateTime.UtcNow;
            state[Side.Buy] = SideState.Idle;
            state[Side.Sell] = SideState.Idle;
        }
