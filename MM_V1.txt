using System;
using System.Linq;
using System.Collections.Generic;
using TradingPlatform.BusinessLayer;

namespace TrueMarketMaking
{
    public class TrueMarketMaking : Strategy
    {
        // ===== Minimal Inputs (MES defaults) =====
        [InputParameter("Symbol", 10)] private Symbol symbol;
        [InputParameter("Account", 20)] public Account account;

        [InputParameter("Contracts (per side)", 30, 1, 1000, 1, 0)]
        public int contracts = 1;

        [InputParameter("Base half-spread (ticks)", 40, 1, 50, 1, 0)]
        public int baseHalfTicks = 2;

        [InputParameter("Queue behind ticks", 41, 0, 10, 1, 0)]
        public int queueBehindTicks = 0;

        [InputParameter("Refresh (ms)", 50, 50, 5000, 50, 0)]
        public int refreshMs = 800;

        [InputParameter("Min rest time (ms)", 60, 0, 20000, 100, 0)]
        public int minOrderAgeMs = 4000;

        [InputParameter("Rest band (ticks)", 61, 1, 10, 1, 0)]
        public int restBandTicks = 3;

        [InputParameter("Force requote (ms, 0=off)", 62, 0, 600000, 100, 0)]
        public int forceRequoteMs = 45000;

        [InputParameter("Ack timeout (ms)", 63, 500, 10000, 100, 0)]
        public int ackTimeoutMs = 4000;

        [InputParameter("Time in force", 64)]
        public TimeInForce timeInForce = TimeInForce.GTC;

        [InputParameter("Use polling ack fallback", 65)]
        public bool usePollingAckFallback = true;

        // Volatility inputs
        [InputParameter("Vol window (sec)", 70, 5, 600, 1, 0)]
        public int volWindowSec = 60;

        [InputParameter("Vol boost factor", 71, 0.0, 5.0, 0.1, 1)]
        public double volBoost = 0.8;

        [InputParameter("Max vol boost (ticks)", 72, 0, 10, 1, 0)]
        public int maxVolBoostTicks = 4;

        // Inventory inputs
        [InputParameter("Enable inventory skew", 80)]
        public bool enableInventory = true;

        [InputParameter("Net position (manual, contracts)", 81, -1000, 1000, 1, 0)]
        public int netPosition = 0;

        [InputParameter("Inventory band (±contracts)", 82, 0, 50, 1, 0)]
        public int invBand = 2;

        [InputParameter("Inventory cap (±contracts)", 83, 1, 100, 1, 0)]
        public int invCap = 4;

        [InputParameter("Max inventory skew (ticks)", 84, 0, 10, 1, 0)]
        public int invMaxSkewTicks = 2;

        [InputParameter("Debug logs", 5)]
        public bool debugLogs = false;

        public override string[] MonitoringConnectionsIds => new[] { this.symbol?.ConnectionId };

        // ===== State machine =====
        private enum SideState { Idle, AwaitingAck, Working, Cancelling }
        private readonly Dictionary<Side, SideState> state = new()
        {
            { Side.Buy,  SideState.Idle },
            { Side.Sell, SideState.Idle }
        };
        private readonly Dictionary<Side, string> orderIdBySide = new() { { Side.Buy, null }, { Side.Sell, null } };
        private readonly Dictionary<Side, DateTime> lastPlaceReqTime = new() { { Side.Buy, DateTime.MinValue }, { Side.Sell, DateTime.MinValue } };
        private readonly Dictionary<Side, DateTime> lastAckTime = new() { { Side.Buy, DateTime.MinValue }, { Side.Sell, DateTime.MinValue } };
        private readonly Dictionary<Side, double> workingPrice = new() { { Side.Buy, double.NaN }, { Side.Sell, double.NaN } };
        private readonly Dictionary<Side, DateTime> lastActionTime = new() { { Side.Buy, DateTime.MinValue }, { Side.Sell, DateTime.MinValue } };

        // ===== Volatility window =====
        private struct MidSample { public DateTime Time; public double Price; }
        private readonly Queue<MidSample> mids = new();

        private DateTime lastLoop = DateTime.MinValue;

        public TrueMarketMaking()
        {
            this.Name = "TrueMarketMaking (MES Pro-Lite)";
            this.Description = "MES market maker: rest-focused, volatility-aware spread, inventory bands.";
        }

        protected override void OnRun()
        {
            if (symbol == null || account == null || symbol.ConnectionId != account.ConnectionId)
            {
                Log("Set Symbol & Account on same connection.", StrategyLoggingLevel.Error);
                return;
            }

            symbol = Core.GetSymbol(symbol.CreateInfo());
            if (symbol == null)
            {
                Log("Symbol not found.", StrategyLoggingLevel.Error);
                return;
            }

            symbol.NewQuote += OnQuote;
            Core.Instance.OrderAdded += OnOrderAdded;
            Core.Instance.OrderRemoved += OnOrderRemoved;

            Log($"Started {symbol.Name} on {account.Name}", StrategyLoggingLevel.Trading);
        }

        protected override void OnStop()
        {
            if (symbol != null) symbol.NewQuote -= OnQuote;
            Core.Instance.OrderAdded -= OnOrderAdded;
            Core.Instance.OrderRemoved -= OnOrderRemoved;

            TryCancel(Side.Buy);
            TryCancel(Side.Sell);
            Log("Stopped", StrategyLoggingLevel.Trading);
        }

        // =================== Main loop ===================
        private void OnQuote(Symbol s, Quote q)
        {
            var now = DateTime.UtcNow;
            if ((now - lastLoop).TotalMilliseconds < refreshMs) return;
            lastLoop = now;

            if (s.Bid <= 0 || s.Ask <= 0) return;

            double tick = s.TickSize > 0 ? s.TickSize : 0.25;
            double mid = (s.Bid + s.Ask) * 0.5;

            PushMid(now, mid, TimeSpan.FromSeconds(Math.Max(5, volWindowSec)));
            int volTicks = EstimateVolatilityTicks(tick);

            int volBoostTicks = Math.Min(maxVolBoostTicks, (int)Math.Ceiling(volTicks * volBoost));
            int halfTicks = Math.Max(1, baseHalfTicks + volBoostTicks);

            // Inventory skew
            int invSkew = 0;
            if (enableInventory && invBand > 0)
            {
                if (Math.Abs(netPosition) >= invCap)
                    invSkew = netPosition > 0 ? +invMaxSkewTicks : -invMaxSkewTicks;
                else
                {
                    int sign = Math.Sign(netPosition);
                    int mag = Math.Max(0, Math.Abs(netPosition) - invBand);
                    invSkew = Math.Clamp(sign * mag, -invMaxSkewTicks, invMaxSkewTicks);
                }
            }

            int bidTicksAdj = +invSkew;
            int askTicksAdj = -invSkew;

            int bidTicks = Math.Max(1, halfTicks + queueBehindTicks + bidTicksAdj);
            int askTicks = Math.Max(1, halfTicks + queueBehindTicks + askTicksAdj);

            double buyPx = RoundToTick(mid - bidTicks * tick, tick);
            double sellPx = RoundToTick(mid + askTicks * tick, tick);

            bool blockBuy = enableInventory && netPosition >= invCap;
            bool blockSell = enableInventory && netPosition <= -invCap;

            StepSide(Side.Buy, buyPx, now, tick, blockBuy);
            StepSide(Side.Sell, sellPx, now, tick, blockSell);

            if (debugLogs)
                Log($"volTicks={volTicks} volBoost={volBoostTicks} invSkew={invSkew} -> bidTicks={bidTicks} askTicks={askTicks}", StrategyLoggingLevel.Trading);
        }

        private void StepSide(Side side, double targetPrice, DateTime now, double tick, bool blockNewPlaces)
        {
            double sinceAck = (now - lastAckTime[side]).TotalMilliseconds;
            double sincePlace = (now - lastPlaceReqTime[side]).TotalMilliseconds;
            double sinceAction = (now - lastActionTime[side]).TotalMilliseconds;

            if (usePollingAckFallback)
            {
                if (state[side] == SideState.AwaitingAck && sincePlace > ackTimeoutMs / 2)
                {
                    var live = FindLiveOnSide(side);
                    if (live != null)
                    {
                        orderIdBySide[side] = live.Id;
                        workingPrice[side] = live.Price;
                        lastAckTime[side] = now;
                        state[side] = SideState.Working;
                    }
                }
            }

            bool forceRequote = (forceRequoteMs > 0) && (sinceAck >= forceRequoteMs);

            switch (state[side])
            {
                case SideState.Idle:
                    if (!blockNewPlaces)
                    {
                        Place(side, targetPrice);
                        state[side] = SideState.AwaitingAck;
                        lastPlaceReqTime[side] = now;
                        lastActionTime[side] = now;
                    }
                    break;

                case SideState.Working:
                    {
                        var cur = workingPrice[side];
                        int deltaTicks = (int)Math.Abs(Math.Round((targetPrice - cur) / tick));
                        bool movedBeyondBand = deltaTicks >= restBandTicks;
                        bool oldEnough = sinceAck >= minOrderAgeMs;

                        if ((movedBeyondBand && oldEnough) || forceRequote)
                        {
                            state[side] = SideState.Cancelling;
                            TryCancel(side);
                            lastAckTime[side] = now;
                            lastActionTime[side] = now;
                        }
                        break;
                    }
            }
        }

        // =================== Trading ops ===================
        private void Place(Side side, double price)
        {
            double step = symbol.LotStep > 0 ? symbol.LotStep : 1.0;
            double minLot = symbol.MinLot > 0 ? symbol.MinLot : 1.0;
            double qty = Math.Max(minLot, Math.Round(contracts / step) * step);

            var req = new PlaceOrderRequestParameters()
            {
                Account = account,
                Symbol = symbol,
                Side = side,
                Price = price,
                Quantity = qty,
                TimeInForce = timeInForce,
                OrderTypeId = OrderType.Limit
            };

            var res = Core.Instance.PlaceOrder(req);
            Log($"Place {side} {qty} @ {price} -> {res}", StrategyLoggingLevel.Trading);
        }

        private void TryCancel(Side side)
        {
            var id = orderIdBySide[side];
            var ord = !string.IsNullOrEmpty(id) ? Core.Instance.GetOrderById(id, symbol.ConnectionId) : null;

            if (ord != null)
            {
                try { Core.Instance.CancelOrder((IOrder)ord); }
                catch { }
                return;
            }

            var arr = Core.Instance.Orders
                .Where(o => o != null && o.Symbol == this.symbol && o.Account == this.account && o.Side == side)
                .ToArray();
            foreach (var o in arr)
                try { Core.Instance.CancelOrder((IOrder)o); } catch { }
        }

        // =================== Events ===================
        private void OnOrderAdded(Order o)
        {
            if (!IsOurs(o)) return;
            var side = o.Side;
            orderIdBySide[side] = o.Id;
            workingPrice[side] = o.Price;
            lastAckTime[side] = DateTime.UtcNow;
            state[side] = SideState.Working;
        }

        private void OnOrderRemoved(Order o)
        {
            if (!IsOurs(o)) return;
            var side = o.Side;
            if (orderIdBySide[side] == o.Id)
            {
                orderIdBySide[side] = null;
                workingPrice[side] = double.NaN;
                lastAckTime[side] = DateTime.UtcNow;
                state[side] = SideState.Idle;
            }
        }

        private bool IsOurs(Order o)
        {
            if (o == null || o.Account == null || o.Symbol == null) return false;
            return o.Account.Id == account.Id && o.Symbol.Id == symbol.Id;
        }

        // ===== Helpers =====
        private static double RoundToTick(double price, double tick) =>
            tick > 0 ? Math.Round(price / tick) * tick : price;

        private Order FindLiveOnSide(Side side) =>
            Core.Instance.Orders.FirstOrDefault(o => o != null && o.Symbol == symbol && o.Account == account && o.Side == side);

        private bool AnyLiveOnSide(Side side) =>
            Core.Instance.Orders.Any(o => o != null && o.Symbol == symbol && o.Account == account && o.Side == side);

        private void PushMid(DateTime t, double price, TimeSpan window)
        {
            mids.Enqueue(new MidSample { Time = t, Price = price });
            var cutoff = t - window;
            while (mids.Count > 0 && mids.Peek().Time < cutoff) mids.Dequeue();
        }

        private int EstimateVolatilityTicks(double tick)
        {
            if (mids.Count < 2 || tick <= 0) return 0;
            double sum = 0; int n = 0;
            MidSample prev = mids.Peek();
            foreach (var x in mids)
            {
                if (x.Time == prev.Time) continue;
                double dticks = Math.Abs((x.Price - prev.Price) / tick);
                sum += dticks; n++;
                prev = x;
            }
            if (n == 0) return 0;
            return (int)Math.Round(sum / n);
        }
    }
}
